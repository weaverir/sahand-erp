// GET - LIST SHIFTS
// ==========================================

export async function GET(request: NextRequest) {
  try {
    await authorizeGlobalRequest(
      request,
      "attendance.shift.view"
    );

    const shifts = await prisma.workShift.findMany({
      include: {
        days: {
          orderBy: [
            { dayOfWeek: "asc" },
            { dayType: "asc" },
            { segmentOrder: "asc" },
          ],
        },

        _count: {
          select: {
            assignments: true,
          },
        },
      },

      orderBy: [
        { isDefault: "desc" },
        { createdAt: "desc" },
      ],
    });

    return successResponse({
      shifts,
      count: shifts.length,
    });

  } catch (error) {
    return getErrorResponse(error);
  }
}
