import { NextRequest } from "next/server";
import { prisma } from "@/libs/prisma";
import { errorResponse, successResponse } from "@/libs/api-response";
import { assertSameOriginRequest, authorizeRequest, RequestOriginError } from "@/modules/auth/auth.middleware";
import { AuthError } from "@/modules/auth/auth.service";
import { PermissionDeniedError } from "@/modules/permissions/access-control";
import { createChallengeNonce, getApiError, getRequestIp, isPlainObject, sha256 } from "@/modules/attendance/attendance-utils";

class ChallengeError extends Error {
  constructor(message: string, public statusCode = 400) { super(message); }
}
function handleError(error: unknown) {
  if (error instanceof ChallengeError || error instanceof RequestOriginError ||
      error instanceof AuthError || error instanceof PermissionDeniedError) {
    return errorResponse(error.message, error.statusCode);
  }
  const parsed = getApiError(error);
  console.error("[ATTENDANCE_CHALLENGE]", error);
  return errorResponse(parsed.message, parsed.statusCode);
}

export async function POST(request: NextRequest) {
  try {
    assertSameOriginRequest(request);
    const auth = await authorizeRequest(request, "attendance.attendance.create");
    if (!auth.assignmentId) throw new ChallengeError("برای این حساب مسئولیت سازمانی فعال تعریف نشده است.", 403);

    const body: unknown = await request.json().catch(() => null);
    if (!isPlainObject(body) || (body.action !== "CHECK_IN" && body.action !== "CHECK_OUT")) {
      throw new ChallengeError("نوع عملیات باید CHECK_IN یا CHECK_OUT باشد.");
    }

    const now = new Date();
    const assignment = await prisma.employeeAssignment.findFirst({
      where: {
        id: auth.assignmentId, userId: auth.user.id, status: "ACTIVE",
        startDate: { lte: now }, OR: [{ endDate: null }, { endDate: { gt: now } }],
      },
      include: { location: true, organizationUnit: { include: { location: true } } },
    });
    if (!assignment) throw new ChallengeError("مسئولیت سازمانی فعال پیدا نشد.", 403);

    const location = assignment.location ?? assignment.organizationUnit.location;
    if (!location || location.status !== "ACTIVE" || !location.geofenceEnabled ||
        location.latitude === null || location.longitude === null) {
      throw new ChallengeError("مختصات یا محدوده حضور محل کار تنظیم نشده است.", 409);
    }

    const faceProfile = await prisma.faceIdentityProfile.findUnique({
      where: { userId: auth.user.id },
      select: { isActive: true, consentAcceptedAt: true },
    });
    if (!faceProfile || !faceProfile.isActive || !faceProfile.consentAcceptedAt) {
      throw new ChallengeError(
        "برای این پرسنل تصویر مرجع و رضایت ثبت‌شده احراز چهره وجود ندارد.",
        409
      );
    }

    const nonce = createChallengeNonce();
    const expiresAt = new Date(now.getTime() + 90_000);
    const challenge = await prisma.attendanceChallenge.create({
      data: {
        userId: auth.user.id,
        employeeAssignmentId: assignment.id,
        action: body.action,
        nonceHash: sha256(nonce),
        expiresAt,
        ipAddress: getRequestIp(request),
        userAgentHash: sha256(request.headers.get("user-agent") ?? ""),
      },
      select: { id: true, action: true, expiresAt: true },
    });

    const response = successResponse({
      challengeId: challenge.id,
      nonce,
      action: challenge.action,
      serverTime: now.toISOString(),
      expiresAt: challenge.expiresAt.toISOString(),
      geofenceRadiusMeters: location.geofenceRadiusMeters,
      message: "چالش ایجاد شد؛ احراز زنده چهره و GPS باید پیش از انقضا تکمیل شوند.",
    }, 201);
    response.headers.set("Cache-Control", "no-store, private");
    return response;
  } catch (error) { return handleError(error); }
}

import {

class VerifyError extends Error {
  constructor(message: string, public statusCode = 400) { super(message); }
}
type VerifyBody = {
  nonce: string; latitude: number; longitude: number;
  accuracyMeters: number; liveSessionToken: string;
};
function parseBody(value: unknown): VerifyBody {
  if (!isPlainObject(value)) throw new VerifyError("اطلاعات درخواست معتبر نیست.");
  const { nonce, latitude, longitude, accuracyMeters, liveSessionToken } = value;
  if (typeof nonce !== "string" || nonce.length < 30 || nonce.length > 100) {
    throw new VerifyError("چالش امنیتی معتبر نیست.");
  }
  if (typeof latitude !== "number" || !Number.isFinite(latitude) || latitude < -90 || latitude > 90 ||
      typeof longitude !== "number" || !Number.isFinite(longitude) || longitude < -180 || longitude > 180) {
    throw new VerifyError("مختصات GPS معتبر نیست.");
  }
  if (typeof accuracyMeters !== "number" || !Number.isFinite(accuracyMeters) ||
      accuracyMeters <= 0 || accuracyMeters > 30) {
    throw new VerifyError("دقت GPS کافی نیست؛ دقت باید حداکثر ۳۰ متر باشد.", 422);
  }
  if (typeof liveSessionToken !== "string" || liveSessionToken.length < 20 || liveSessionToken.length > 5000) {
    throw new VerifyError("توکن نشست زنده چهره معتبر نیست.");
  }
  return { nonce, latitude, longitude, accuracyMeters, liveSessionToken };
}
async function failChallenge(id: string, reason: string) {
  await prisma.attendanceChallenge.updateMany({
    where: { id, status: AttendanceChallengeStatus.OPEN, consumedAt: null },
    data: { status: AttendanceChallengeStatus.FAILED, failedAt: new Date(), failureReason: reason.slice(0, 200) },
  });
}
function handleError(error: unknown) {
  if (error instanceof VerifyError || error instanceof RequestOriginError ||
      error instanceof AuthError || error instanceof PermissionDeniedError ||
      error instanceof FaceProviderUnavailableError || error instanceof FaceVerificationRejectedError) {
    return errorResponse(error.message, error.statusCode);
  }
  const parsed = getApiError(error);
  console.error("[ATTENDANCE_VERIFY]", error);
  return errorResponse(parsed.message, parsed.statusCode);
}

export async function POST(
  request: NextRequest,
  context: { params: Promise<{ challengeId: string }> }
) {
  let challengeId = "";
  try {
    assertSameOriginRequest(request);
    challengeId = (await context.params).challengeId;
    if (!challengeId || challengeId.length > 100) throw new VerifyError("شناسه چالش معتبر نیست.", 404);

    const challenge = await prisma.attendanceChallenge.findUnique({ where: { id: challengeId } });
    if (!challenge) throw new VerifyError("چالش پیدا نشد.", 404);

    const auth = await authorizeRequest(request, "attendance.attendance.create", challenge.employeeAssignmentId);
    if (challenge.userId !== auth.user.id) throw new VerifyError("چالش پیدا نشد.", 404);
    if (challenge.status !== AttendanceChallengeStatus.OPEN || challenge.consumedAt) {
      throw new VerifyError("چالش قبلاً استفاده شده یا معتبر نیست.", 409);
    }

    const now = new Date();
    if (challenge.expiresAt <= now) {
      await prisma.attendanceChallenge.updateMany({
        where: { id: challenge.id, status: AttendanceChallengeStatus.OPEN },
        data: { status: AttendanceChallengeStatus.EXPIRED, failureReason: "CHALLENGE_EXPIRED" },
      });
      throw new VerifyError("مهلت درخواست تمام شده است؛ دوباره شروع کنید.", 410);
    }

    const body = parseBody(await request.json().catch(() => null));
    if (!safeHexEqual(challenge.nonceHash, sha256(body.nonce))) {
      await failChallenge(challenge.id, "INVALID_NONCE");
      throw new VerifyError("اعتبار چالش امنیتی تأیید نشد.", 403);
    }

    const assignment = await prisma.employeeAssignment.findFirst({
      where: {
        id: challenge.employeeAssignmentId, userId: auth.user.id, status: "ACTIVE",
        startDate: { lte: now }, OR: [{ endDate: null }, { endDate: { gt: now } }],
      },
      include: { location: true, organizationUnit: { include: { location: true } } },
    });
    if (!assignment) {
      await failChallenge(challenge.id, "ASSIGNMENT_INACTIVE");
      throw new VerifyError("مسئولیت سازمانی فعال نیست.", 403);
    }

    const location = assignment.location ?? assignment.organizationUnit.location;
    if (!location || location.status !== "ACTIVE" || !location.geofenceEnabled ||
        location.latitude === null || location.longitude === null) {
      await failChallenge(challenge.id, "LOCATION_NOT_CONFIGURED");
      throw new VerifyError("محدوده حضور معتبر پیکربندی نشده است.", 409);
    }

    const distance = calculateDistanceMeters(
      body.latitude, body.longitude, Number(location.latitude), Number(location.longitude)
    );
    if (distance + body.accuracyMeters > location.geofenceRadiusMeters) {
      await failChallenge(challenge.id, "OUTSIDE_OR_UNCERTAIN_GEOFENCE");
      throw new VerifyError(
        `موقعیت با احتساب خطای GPS داخل محدوده مطمئن ۵۰ متری نیست؛ فاصله ${Math.round(distance)} متر و خطای اعلامی ${Math.round(body.accuracyMeters)} متر است.`,
        403
      );
    }

    const faceProfile = await prisma.faceIdentityProfile.findUnique({ where: { userId: auth.user.id } });
    if (!faceProfile || !faceProfile.isActive || !faceProfile.consentAcceptedAt) {
      throw new VerifyError("تصویر مرجع یا رضایت ثبت‌شده برای احراز چهره وجود ندارد.", 409);
    }

    let referenceImage: Buffer;
    try {
      referenceImage = decryptReferenceImage({
        encrypted: faceProfile.referenceImageEncrypted,
        iv: faceProfile.referenceImageIv,
        authTag: faceProfile.referenceImageAuthTag,
        keyVersion: faceProfile.encryptionKeyVersion,
      });
    } catch {
      throw new VerifyError("بازیابی امن تصویر مرجع ممکن نیست؛ با منابع انسانی تماس بگیرید.", 503);
    }

    const face = await verifyLiveFace({
      challengeId: challenge.id, challengeNonce: body.nonce,
      userId: auth.user.id, action: challenge.action,
      liveSessionToken: body.liveSessionToken,
      referenceImage, referenceImageMimeType: faceProfile.referenceImageMimeType,
    });

    const workDate = getWorkDate(now);
    const dayOfWeek = getSaturdayBasedDayOfWeek(workDate);
    const shiftAssignments = await prisma.shiftAssignment.findMany({
      where: {
        isActive: true, validFrom: { lte: workDate },
        AND: [{ OR: [{ validTo: null }, { validTo: { gte: workDate } }] }],
        OR: [
          { targetType: "EMPLOYEE", employeeAssignmentId: assignment.id },
          { targetType: "UNIT", organizationUnitId: assignment.organizationUnitId },
          { targetType: "COMPANY" },
        ],
      },
      include: { shift: { include: { days: true } } },
    });
    const rank = (target: string) => target === "EMPLOYEE" ? 3 : target === "UNIT" ? 2 : target === "COMPANY" ? 1 : 0;
    shiftAssignments.sort((a, b) => rank(b.targetType) - rank(a.targetType));
    const selectedShift = shiftAssignments[0] ?? null;
    const segments = selectedShift?.shift.days.filter(
      day => day.dayOfWeek === dayOfWeek && day.dayType === "NORMAL"
    ) ?? [];
    const requiredMinutes = segments.length
      ? segments.reduce((sum, day) => sum + (day.isWorkingDay ? day.requiredMinutes : 0), 0)
      : 480;

    const eventType = challenge.action === "CHECK_IN"
      ? AttendanceEventType.CHECK_IN : AttendanceEventType.CHECK_OUT;

    const saved = await prisma.$transaction(async tx => {
      const consumed = await tx.attendanceChallenge.updateMany({
        where: {
          id: challenge.id, userId: auth.user.id,
          status: AttendanceChallengeStatus.OPEN, consumedAt: null,
          expiresAt: { gt: now },
        },
        data: { status: AttendanceChallengeStatus.CONSUMED, consumedAt: now },
      });
      if (consumed.count !== 1) throw new VerifyError("چالش قبلاً مصرف شده یا منقضی شده است.", 409);

      let record = await tx.attendanceRecord.findUnique({
        where: { employeeAssignmentId_workDate: { employeeAssignmentId: assignment.id, workDate } },
      });
      if (!record) {
        record = await tx.attendanceRecord.create({
          data: {
            employeeAssignmentId: assignment.id,
            shiftAssignmentId: selectedShift?.id ?? null,
            checkpointLocationId: location.id,
            workDate,
            timeZone: process.env.ATTENDANCE_TIME_ZONE || "Asia/Tehran",
            requiredMinutes,
            status: "OPEN",
          },
        });
      }

      const latest = await tx.attendanceEvent.findFirst({
        where: { attendanceRecordId: record.id },
        orderBy: [{ occurredAt: "desc" }, { createdAt: "desc" }],
      });
      if (challenge.action === "CHECK_IN") {
        if (latest && latest.eventType !== AttendanceEventType.CHECK_OUT &&
            latest.eventType !== AttendanceEventType.AUTO_CHECK_OUT) {
          throw new VerifyError("ورود قبلی هنوز بسته نشده است.", 409);
        }
      } else if (!latest || (
        latest.eventType !== AttendanceEventType.CHECK_IN &&
        latest.eventType !== AttendanceEventType.BREAK_END
      )) {
        throw new VerifyError("برای ثبت خروج، ابتدا باید ورود معتبر ثبت شده باشد.", 409);
      }

      const event = await tx.attendanceEvent.create({
        data: {
          attendanceRecordId: record.id, eventType, occurredAt: now,
          latitude: body.latitude, longitude: body.longitude,
          accuracyMeters: body.accuracyMeters,
          distanceFromCheckpointMeters: distance, geofencePassed: true,
          source: "PWA", ipAddress: getRequestIp(request),
          userAgent: request.headers.get("user-agent"),
          idempotencyKey: challenge.id,
          notes: `Face provider: ${face.provider}; verification: ${face.verificationId}`,
        },
      });

      await tx.attendanceVerification.createMany({
        data: [
          {
            attendanceEventId: event.id, verificationType: AttendanceVerificationType.GPS,
            status: AttendanceVerificationStatus.PASSED, provider: "BROWSER_GEOLOCATION",
            details: { distanceMeters: distance, accuracyMeters: body.accuracyMeters,
              geofenceRadiusMeters: location.geofenceRadiusMeters }, verifiedAt: now,
          },
          {
            attendanceEventId: event.id, verificationType: AttendanceVerificationType.FACE,
            status: AttendanceVerificationStatus.PASSED, score: face.similarityScore,
            provider: face.provider, details: { verificationId: face.verificationId }, verifiedAt: now,
          },
          {
            attendanceEventId: event.id, verificationType: AttendanceVerificationType.LIVENESS,
            status: AttendanceVerificationStatus.PASSED, provider: face.provider,
            details: { verificationId: face.verificationId }, verifiedAt: now,
          },
        ],
      });

      const allEvents = await tx.attendanceEvent.findMany({
        where: { attendanceRecordId: record.id },
        orderBy: { occurredAt: "asc" },
        select: { eventType: true, occurredAt: true },
      });
      let openAt: Date | null = null, workedMinutes = 0;
      let firstCheckInAt: Date | null = null, lastCheckOutAt: Date | null = null;
      for (const item of allEvents) {
        if (item.eventType === AttendanceEventType.CHECK_IN) {
          firstCheckInAt ??= item.occurredAt;
          openAt = item.occurredAt;
        } else if (item.eventType === AttendanceEventType.CHECK_OUT ||
                   item.eventType === AttendanceEventType.AUTO_CHECK_OUT) {
          lastCheckOutAt = item.occurredAt;
          if (openAt) {
            workedMinutes += Math.max(0, Math.floor((item.occurredAt.getTime() - openAt.getTime()) / 60000));
            openAt = null;
          }
        }
      }

      const updatedRecord = await tx.attendanceRecord.update({
        where: { id: record.id },
        data: {
          firstCheckInAt, lastCheckOutAt, workedMinutes,
          overtimeMinutes: Math.max(workedMinutes - requiredMinutes, 0),
          status: challenge.action === "CHECK_IN" ? "OPEN" : "COMPLETED",
        },
      });
      return { event, record: updatedRecord };
    }, { isolationLevel: "Serializable" });

    const response = successResponse({
      event: saved.event, record: saved.record,
      message: challenge.action === "CHECK_IN" ? "ورود با موفقیت ثبت شد." : "خروج با موفقیت ثبت شد.",
    });
    response.headers.set("Cache-Control", "no-store, private");
    return response;
  } catch (error) {
    if (challengeId && (
      error instanceof FaceVerificationRejectedError ||
      error instanceof VerifyError
    )) {
      await failChallenge(challengeId, error.message).catch(() => undefined);
    }
    return handleError(error);
  }
}

import { createHash, timingSafeEqual } from "node:crypto";

export async function POST(request: NextRequest) {
  try {
    assertSameOriginRequest(request);
    const user = await authenticateRequest(request);
    const body = await request.json();
    if (typeof body?.challengeId !== "string" || typeof body?.nonce !== "string")
      return Response.json({ error: "درخواست نامعتبر است." }, { status: 400 });

    if (!challenge || challenge.userId !== user.id || challenge.status !== "OPEN" ||
        challenge.consumedAt || challenge.expiresAt <= new Date())
      return Response.json({ error: "چالش معتبر نیست." }, { status: 403 });
    const digest = createHash("sha256").update(body.nonce).digest();

      return Response.json({ error: "چالش معتبر نیست." }, { status: 403 });

      challengeId: challenge.id, challengeNonce: body.nonce,
      subjectId: user.id, action: challenge.action,
    });
    return Response.json(data, { headers: { "cache-control": "no-store" } });
  } catch {
    return Response.json({ error: "ایجاد نشست داخلی ناموفق بود." }, { status: 503 });
  }
}

import "server-only";

const base = process.env.LOCAL_FACE_SERVICE_URL ?? "http://127.0.0.1:8765";
const secret = process.env.LOCAL_FACE_SERVICE_KEY;

export async function localFaceRequest<T>(path: string, payload: object): Promise<T> {
  if (!secret || secret.length < 32) throw new Error("LOCAL_FACE_SERVICE_KEY تنظیم نشده است.");
  // Private loopback by default; remote deployment requires a protected internal network.
  const endpoint = new URL(path, base);
  if (!(["127.0.0.1", "localhost", "::1"].includes(endpoint.hostname) ||
    process.env.LOCAL_FACE_ALLOW_PRIVATE_NETWORK === "true")) {
    throw new Error("آدرس سرویس چهره باید داخلی و قابل اعتماد باشد.");
  }
  const response = await fetch(endpoint, {
    method: "POST", cache: "no-store",
    headers: { "content-type": "application/json", authorization: `Bearer ${secret}` },
    body: JSON.stringify(payload), signal: AbortSignal.timeout(15000),
  });
  if (!response.ok) throw new Error(`سرویس داخلی چهره درخواست را رد کرد (${response.status}).`);
  return response.json() as Promise<T>;
}
