export class RequestOriginError extends Error {
  statusCode = 403;

  constructor(
    message = "مبدأ درخواست مجاز نیست."
  ) {
    super(message);
    this.name = "RequestOriginError";
  }
}

/**
 * بررسی مبدأ درخواست‌های تغییردهنده اطلاعات
 */
export function assertSameOriginRequest(
  request: NextRequest
) {
  const origin = request.headers.get("origin");
  const referer = request.headers.get("referer");

  const source = origin || referer;

  if (!source) {
    throw new RequestOriginError(
      "مبدأ درخواست مشخص نیست."
    );
  }

  let sourceOrigin: string;

  try {
    sourceOrigin = new URL(source).origin;
  } catch {
    throw new RequestOriginError();
  }

  if (sourceOrigin !== request.nextUrl.origin) {
    throw new RequestOriginError();
  }

  const fetchSite = request.headers.get("sec-fetch-site");

  if (fetchSite === "cross-site") {
    throw new RequestOriginError();
  }
}

/**
 * احراز هویت کاربر
 */
export async function authenticateRequest(
  request: NextRequest
) {
  const accessToken = request.cookies.get(
    AUTH_COOKIE_NAMES.access
  )?.value;

  if (!accessToken) {
    throw new AuthError(
      "ابتدا وارد حساب کاربری شوید."
    );
  }

  return getAuthenticatedUser(accessToken);
}

/**
 * بررسی مجوز با در نظر گرفتن مسئولیت فعال سازمانی
 */
export async function authorizeRequest(
  request: NextRequest,
  permissionKey: string,
  assignmentId?: string
) {
  const user = await authenticateRequest(request);

  // دریافت مسئولیت انتخاب‌شده از هدر درخواست
  const headerAssignmentId = request.headers.get(
    "x-organization-assignment-id"
  )?.trim();

  const selectedAssignmentId =
    assignmentId ?? (headerAssignmentId || undefined);

  const permission = await checkPermission({
    userId: user.id,
    permission: permissionKey,
    assignmentId: selectedAssignmentId,
  });

  if (!permission.allowed) {
    throw new PermissionDeniedError();
  }

  return {
    user,
    permission,
    assignmentId: permission.activeAssignmentId,
  };
}

/**
 * بررسی دسترسی سراسری به اطلاعات مرجع
 */
export async function authorizeGlobalRequest(
  request: NextRequest,
  permissionKey: string
) {
  const authorization = await authorizeRequest(
    request,
    permissionKey
  );

  if (
    !authorization.permission.scopes.includes(
      DataScope.ALL
    )
  ) {
    throw new PermissionDeniedError(
      "برای دسترسی به اطلاعات مرجع، مجوز سراسری لازم است."
    );
  }

  return authorization;
}

import "server-only";

import { prisma } from "@/libs/prisma";
import { DataScope } from "@/generated/prisma/client";

type PermissionCheck = {
  userId: string;
  permission: string;
  assignmentId?: string;
};

export type PermissionResult = {
  allowed: boolean;
  scopes: DataScope[];
  isSystemAdmin: boolean;
  activeAssignmentId: string | null;
};

export class PermissionDeniedError extends Error {
  statusCode = 403;

  constructor(
    message = "شما اجازه انجام این عملیات را ندارید."
  ) {
    super(message);
    this.name = "PermissionDeniedError";
  }
}

const emptyResult = (): PermissionResult => ({
  allowed: false,
  scopes: [],
  isSystemAdmin: false,
  activeAssignmentId: null,
});

/**
 * بررسی مجوز در سطح کاربر و مسئولیت سازمانی
 *
 * اگر assignmentId ارسال نشود:
 * فقط مسئولیت اصلی و فعال کاربر بررسی می‌شود.
 *
 * اگر assignmentId ارسال شود:
 * مالکیت و فعال بودن آن مسئولیت بررسی می‌شود.
 */
export async function checkPermission({
  userId,
  permission,
  assignmentId,
}: PermissionCheck): Promise<PermissionResult> {
  const now = new Date();

  const user = await prisma.user.findFirst({
    where: {
      id: userId,
      isActive: true,
    },

    select: {
      id: true,

      roles: {
        where: {
          isActive: true,

          OR: [
            { expiresAt: null },
            { expiresAt: { gt: now } },
          ],

          role: {
            isActive: true,
          },
        },

        select: {
          role: {
            select: {
              key: true,
              isSystem: true,

              permissions: {
                where: {
                  permission: {
                    key: permission,
                  },
                },

                select: {
                  isAllowed: true,
                  scope: true,
                },
              },
            },
          },
        },
      },

      employeeAssignments: {
        where: {
          status: "ACTIVE",

          AND: [
            {
              OR: [
                { endDate: null },
                { endDate: { gt: now } },
              ],
            },

            assignmentId
              ? { id: assignmentId }
              : { isPrimary: true },
          ],
        },

        orderBy: [
          { isPrimary: "desc" },
          { startDate: "desc" },
        ],

        take: 1,

        select: {
          id: true,

          roles: {
            where: {
              isActive: true,

              OR: [
                { expiresAt: null },
                { expiresAt: { gt: now } },
              ],

              role: {
                isActive: true,
                isSystem: false,
              },
            },

            select: {
              role: {
                select: {
                  permissions: {
                    where: {
                      permission: {
                        key: permission,
                      },
                    },

                    select: {
                      isAllowed: true,
                      scope: true,
                    },
                  },
                },
              },
            },
          },

          permissions: {
            where: {
              isActive: true,

              OR: [
                { expiresAt: null },
                { expiresAt: { gt: now } },
              ],

              permission: {
                key: permission,
              },
            },

            select: {
              isAllowed: true,
              scope: true,
            },
          },
        },
      },
    },
  });

  if (!user) {
    return emptyResult();
  }

  const assignment =
    user.employeeAssignments[0] ?? null;

  // مسئولیت ارسالی باید متعلق به همین کاربر و فعال باشد.
  if (assignmentId && !assignment) {
    return emptyResult();
  }

  // مدیر سیستم به تمام مجوزها دسترسی دارد.
  const isSystemAdmin = user.roles.some(
    (userRole) =>
      userRole.role.isSystem &&
      userRole.role.key === "system_admin"
  );

  if (isSystemAdmin) {
    return {
      allowed: true,
      scopes: [DataScope.ALL],
      isSystemAdmin: true,
      activeAssignmentId: assignment?.id ?? null,
    };
  }

  // مجوزهای نقش‌های عمومی کاربر
  const globalGrants = user.roles.flatMap(
    (userRole) => userRole.role.permissions
  );

  // مجوزهای نقش‌های مسئولیت فعال
  const assignmentRoleGrants =
    assignment?.roles.flatMap(
      (assignmentRole) =>
        assignmentRole.role.permissions
    ) ?? [];

  // مجوزهای مستقیم مسئولیت
  const assignmentDirectGrants =
    assignment?.permissions ?? [];

  const grants = [
    ...globalGrants,
    ...assignmentRoleGrants,
    ...assignmentDirectGrants,
  ];

  // منع صریح دسترسی بر اجازه اولویت دارد.
  if (grants.some((grant) => !grant.isAllowed)) {
    return {
      allowed: false,
      scopes: [],
      isSystemAdmin: false,
      activeAssignmentId: assignment?.id ?? null,
    };
  }

  const allowedGrants = grants.filter(
    (grant) => grant.isAllowed
  );

  if (allowedGrants.length === 0) {
    return emptyResult();
  }

  const scopes = [
    ...new Set(
      allowedGrants.map((grant) => grant.scope)
    ),
  ];

  return {
    allowed: true,
    scopes,
    isSystemAdmin: false,
    activeAssignmentId: assignment?.id ?? null,
  };
}

/**
 * بررسی ساده مجوز
 */
export async function hasPermission(
  userId: string,
  permission: string,
  assignmentId?: string
): Promise<boolean> {
  const result = await checkPermission({
    userId,
    permission,
    assignmentId,
  });

  return result.allowed;
}

/**
 * بررسی اجباری مجوز
 */
export async function requirePermission(
  userId: string,
  permission: string,
  assignmentId?: string
): Promise<PermissionResult> {
  const result = await checkPermission({
    userId,
    permission,
    assignmentId,
  });

  if (!result.allowed) {
    throw new PermissionDeniedError();
  }

  return result;
}

/**
 * بررسی دسترسی به اطلاعات متعلق به کاربر
 */
export function canAccessOwnData(
  result: PermissionResult,
  ownerUserId: string,
  currentUserId: string
): boolean {
  if (!result.allowed) {
    return false;
  }

  if (result.scopes.includes(DataScope.ALL)) {
    return true;
  }

  if (result.scopes.includes(DataScope.OWN)) {
    return ownerUserId === currentUserId;
  }

  return false;
}

              isAllowed: true,
              permission: {
                select: {

                },
              },

      },
    },
  });

  if (!user) {
    return [];
  }

  // مدیر سیستم تمام مجوزها را دارد.
  const isSystemAdmin = user.roles.some(
    (item) =>

  const allowedPermissions = new Set(
    grants
      .filter((item) => item.isAllowed)
      .map((item) => item.permission.key)
  );

  return [...allowedPermissions].filter(
    (key) => !deniedPermissions.has(key)
  );
}

class ShiftError extends Error {
  constructor(
    message: string,
    public statusCode = 400
  ) {
    super(message);
    this.name = "ShiftError";
  }
}

type ShiftDayInput = {
  dayOfWeek: number;
  segmentOrder: number;
  dayType: WorkShiftDayType;
  isWorkingDay: boolean;
  startMinute: number | null;
  endMinute: number | null;
  requiredMinutes: number;
};

// ==========================================
// HELPERS
// ==========================================

function isObject(
  value: unknown
): value is Record<string, unknown> {
  return (
    typeof value === "object" &&
    value !== null &&
    !Array.isArray(value)
  );
}

function readText(
  value: unknown,
  field: string,
  maxLength: number
): string {
  if (typeof value !== "string") {
    throw new ShiftError(`${field} الزامی است.`);
  }

  const result = value.trim();

  if (!result || result.length > maxLength) {
    throw new ShiftError(`${field} معتبر نیست.`);
  }

  return result;
}

function readInteger(
  value: unknown,
  field: string,
  min: number,
  max: number,
  fallback?: number
): number {
  if (value === undefined && fallback !== undefined) {
    return fallback;
  }

  if (
    typeof value !== "number" ||
    !Number.isInteger(value) ||
    value < min ||
    value > max
  ) {
    throw new ShiftError(`${field} معتبر نیست.`);
  }

  return value;
}

function parseDays(value: unknown): ShiftDayInput[] {
  if (!Array.isArray(value) || value.length === 0) {
    throw new ShiftError(
      "حداقل یک روز کاری یا تعطیل باید تعریف شود."
    );
  }

  const allowedTypes = Object.values(
    WorkShiftDayType
  ) as string[];

  const days = value.map((item, index) => {
    if (!isObject(item)) {
      throw new ShiftError("ساختار روز شیفت معتبر نیست.");
    }

    const dayOfWeek = readInteger(
      item.dayOfWeek,
      "روز هفته",
      0,
      6
    );

    const segmentOrder = readInteger(
      item.segmentOrder,
      "ترتیب بازه",
      1,
      100,
      1
    );

    const dayType =
      item.dayType === undefined
        ? WorkShiftDayType.NORMAL
        : item.dayType;

    if (
      typeof dayType !== "string" ||
      !allowedTypes.includes(dayType)
    ) {
      throw new ShiftError("نوع روز معتبر نیست.");
    }

    const isWorkingDay =
      item.isWorkingDay === undefined
        ? true
        : item.isWorkingDay;

    if (typeof isWorkingDay !== "boolean") {
      throw new ShiftError("وضعیت روز کاری معتبر نیست.");
    }

    const startMinute =
      item.startMinute === null ||
      item.startMinute === undefined
        ? null
        : readInteger(
            item.startMinute,
            "ساعت شروع",
            0,
            1439
          );

    const endMinute =
      item.endMinute === null ||
      item.endMinute === undefined
        ? null
        : readInteger(
            item.endMinute,
            "ساعت پایان",
            0,
            1439
          );

    if (
      isWorkingDay &&
      (startMinute === null || endMinute === null)
    ) {
      throw new ShiftError(
        "برای روز کاری ساعت شروع و پایان الزامی است."
      );
    }

    const requiredMinutes = readInteger(
      item.requiredMinutes,
      "ساعت موظفی",
      0,
      1440,
      isWorkingDay ? 480 : 0
    );

    if (!isWorkingDay && requiredMinutes !== 0) {
      throw new ShiftError(
        "روز غیرکاری نباید ساعت موظفی داشته باشد."
      );
    }

    return {
      dayOfWeek,
      segmentOrder,
      dayType: dayType as WorkShiftDayType,
      isWorkingDay,
      startMinute,
      endMinute,
      requiredMinutes,
    };
  });

  const uniqueSegments = new Set(
    days.map(
      (day) =>
        `${day.dayOfWeek}-${day.dayType}-${day.segmentOrder}`
    )
  );

  if (uniqueSegments.size !== days.length) {
    throw new ShiftError(
      "ترتیب بازه‌های یک روز نباید تکراری باشد."
    );
  }

  return days;
}

function getErrorResponse(error: unknown) {
  if (
    error instanceof ShiftError ||
    error instanceof RequestOriginError ||
    error instanceof AuthError ||
    error instanceof PermissionDeniedError
  ) {
    return errorResponse(
      error.message,
      error.statusCode
    );
  }

  if (
    isObject(error) &&
    error.code === "P2002"
  ) {
    return errorResponse(
      "کد این شیفت قبلاً ثبت شده است.",
      409
    );
  }

  console.error("[WORK_SHIFT_ERROR]", error);

  return errorResponse(
    "خطا در پردازش اطلاعات شیفت.",
    500
  );
}

export async function GET(request: NextRequest) {
  try {
    await authorizeGlobalRequest(
      request,
      "attendance.shift.view"
    );

    const shifts = await prisma.workShift.findMany({
      include: {
        days: {
          orderBy: [
            { dayOfWeek: "asc" },
            { dayType: "asc" },
            { segmentOrder: "asc" },
          ],
        },

        _count: {
          select: {
            assignments: true,
          },
        },
      },

      orderBy: [
        { isDefault: "desc" },
        { createdAt: "desc" },
      ],
    });

    return successResponse({
      shifts,
      count: shifts.length,
    });

  } catch (error) {
    return getErrorResponse(error);
  }
}

// POST - CREATE SHIFT
// ==========================================

export async function POST(request: NextRequest) {
  try {
    assertSameOriginRequest(request);

    const { user: actor } =
      await authorizeGlobalRequest(
        request,
        "attendance.shift.create"
      );

    let body: unknown;

    try {
      body = await request.json();
    } catch {
      throw new ShiftError(
        "ساختار درخواست معتبر نیست."
      );
    }

    if (!isObject(body)) {
      throw new ShiftError(
        "اطلاعات شیفت معتبر نیست."
      );
    }

    const code = readText(
      body.code,
      "کد شیفت",
      50
    );

    const name = readText(
      body.name,
      "نام شیفت",
      120
    );

    const description =
      body.description === undefined ||
      body.description === null ||
      body.description === ""
        ? null
        : readText(
            body.description,
            "توضیحات",
            2000
          );

    const isDefault =
      body.isDefault === undefined
        ? false
        : body.isDefault;

    if (typeof isDefault !== "boolean") {
      throw new ShiftError(
        "وضعیت شیفت پیش‌فرض معتبر نیست."
      );
    }

    const days = parseDays(body.days);

    const created = await prisma.$transaction(
      async (tx) => {
        // فقط یک شیفت پیش‌فرض فعال باشد.
        if (isDefault) {
          await tx.workShift.updateMany({
            where: {
              isDefault: true,
            },
            data: {
              isDefault: false,
            },
          });
        }

        const shift = await tx.workShift.create({
          data: {
            code,
            name,
            description,
            isDefault,
            isActive: true,
            version: 1,

            days: {
              create: days,
            },
          },

          include: {
            days: {
              orderBy: [
                { dayOfWeek: "asc" },
                { dayType: "asc" },
                { segmentOrder: "asc" },
              ],
            },
          },
        });

        await tx.organizationAuditLog.create({
          data: {
            actorUserId: actor.id,
            entityType: "WorkShift",
            entityId: shift.id,
            action: "WORK_SHIFT_CREATED",
            ipAddress:
              request.headers
                .get("x-forwarded-for")
                ?.split(",")[0]
                ?.trim() ??
              request.headers.get("x-real-ip"),
            afterData: {
              code,
              name,
              version: shift.version,
              isDefault,
              daysCount: days.length,
            },
          },
        });

        return shift;
      }
    );

    return successResponse(
      {
        shift: created,
        message: "شیفت کاری با موفقیت ثبت شد.",
      },
      201
    );

  } catch (error) {
    return getErrorResponse(error);
  }
}
