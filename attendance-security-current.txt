import "server-only";

import { NextRequest } from "next/server";
import { DataScope } from "@/generated/prisma/client";

import {
  AuthError,
  getAuthenticatedUser,
} from "./auth.service";

import { AUTH_COOKIE_NAMES } from "./auth.cookies";

import {
  checkPermission,
  PermissionDeniedError,
} from "@/modules/permissions/access-control";

export class RequestOriginError extends Error {
  statusCode = 403;

  constructor(
    message = "مبدأ درخواست مجاز نیست."
  ) {
    super(message);
    this.name = "RequestOriginError";
  }
}

/**
 * بررسی مبدأ درخواست‌های تغییردهنده اطلاعات
 */
export function assertSameOriginRequest(
  request: NextRequest
) {
  const origin = request.headers.get("origin");
  const referer = request.headers.get("referer");

  const source = origin || referer;

  if (!source) {
    throw new RequestOriginError(
      "مبدأ درخواست مشخص نیست."
    );
  }

  let sourceOrigin: string;

  try {
    sourceOrigin = new URL(source).origin;
  } catch {
    throw new RequestOriginError();
  }

  if (sourceOrigin !== request.nextUrl.origin) {
    throw new RequestOriginError();
  }

  const fetchSite = request.headers.get("sec-fetch-site");

  if (fetchSite === "cross-site") {
    throw new RequestOriginError();
  }
}

/**
 * احراز هویت کاربر
 */
export async function authenticateRequest(
  request: NextRequest
) {
  const accessToken = request.cookies.get(
    AUTH_COOKIE_NAMES.access
  )?.value;

  if (!accessToken) {
    throw new AuthError(
      "ابتدا وارد حساب کاربری شوید."
    );
  }

  return getAuthenticatedUser(accessToken);
}

/**
 * بررسی مجوز با در نظر گرفتن مسئولیت فعال سازمانی
 */
export async function authorizeRequest(
  request: NextRequest,
  permissionKey: string,
  assignmentId?: string
) {
  const user = await authenticateRequest(request);

  // دریافت مسئولیت انتخاب‌شده از هدر درخواست
  const headerAssignmentId = request.headers.get(
    "x-organization-assignment-id"
  )?.trim();

  const selectedAssignmentId =
    assignmentId ?? (headerAssignmentId || undefined);

  const permission = await checkPermission({
    userId: user.id,
    permission: permissionKey,
    assignmentId: selectedAssignmentId,
  });

  if (!permission.allowed) {
    throw new PermissionDeniedError();
  }

  return {
    user,
    permission,
    assignmentId: permission.activeAssignmentId,
  };
}

/**
 * بررسی دسترسی سراسری به اطلاعات مرجع
 */
export async function authorizeGlobalRequest(
  request: NextRequest,
  permissionKey: string
) {
  const authorization = await authorizeRequest(
    request,
    permissionKey
  );

  if (
    !authorization.permission.scopes.includes(
      DataScope.ALL
    )
  ) {
    throw new PermissionDeniedError(
      "برای دسترسی به اطلاعات مرجع، مجوز سراسری لازم است."
    );
  }

  return authorization;
}

===== ACCESS CONTROL =====
import "server-only";

import { prisma } from "@/libs/prisma";
import { DataScope } from "@/generated/prisma/client";

type PermissionCheck = {
  userId: string;
  permission: string;
  assignmentId?: string;
};

export type PermissionResult = {
  allowed: boolean;
  scopes: DataScope[];
  isSystemAdmin: boolean;
  activeAssignmentId: string | null;
};

export class PermissionDeniedError extends Error {
  statusCode = 403;

  constructor(
    message = "شما اجازه انجام این عملیات را ندارید."
  ) {
    super(message);
    this.name = "PermissionDeniedError";
  }
}

const emptyResult = (): PermissionResult => ({
  allowed: false,
  scopes: [],
  isSystemAdmin: false,
  activeAssignmentId: null,
});

/**
 * بررسی مجوز در سطح کاربر و مسئولیت سازمانی
 *
 * اگر assignmentId ارسال نشود:
 * فقط مسئولیت اصلی و فعال کاربر بررسی می‌شود.
 *
 * اگر assignmentId ارسال شود:
 * مالکیت و فعال بودن آن مسئولیت بررسی می‌شود.
 */
export async function checkPermission({
  userId,
  permission,
  assignmentId,
}: PermissionCheck): Promise<PermissionResult> {
  const now = new Date();

  const user = await prisma.user.findFirst({
    where: {
      id: userId,
      isActive: true,
    },

    select: {
      id: true,

      roles: {
        where: {
          isActive: true,

          OR: [
            { expiresAt: null },
            { expiresAt: { gt: now } },
          ],

          role: {
            isActive: true,
          },
        },

        select: {
          role: {
            select: {
              key: true,
              isSystem: true,

              permissions: {
                where: {
                  permission: {
                    key: permission,
                  },
                },

                select: {
                  isAllowed: true,
                  scope: true,
                },
              },
            },
          },
        },
      },

      employeeAssignments: {
        where: {
          status: "ACTIVE",

          AND: [
            {
              OR: [
                { endDate: null },
                { endDate: { gt: now } },
              ],
            },

            assignmentId
              ? { id: assignmentId }
              : { isPrimary: true },
          ],
        },

        orderBy: [
          { isPrimary: "desc" },
          { startDate: "desc" },
        ],

        take: 1,

        select: {
          id: true,

          roles: {
            where: {
              isActive: true,

              OR: [
                { expiresAt: null },
                { expiresAt: { gt: now } },
              ],

              role: {
                isActive: true,
                isSystem: false,
              },
            },

            select: {
              role: {
                select: {
                  permissions: {
                    where: {
                      permission: {
                        key: permission,
                      },
                    },

                    select: {
                      isAllowed: true,
                      scope: true,
                    },
                  },
                },
              },
            },
          },

          permissions: {
            where: {
              isActive: true,

              OR: [
                { expiresAt: null },
                { expiresAt: { gt: now } },
              ],

              permission: {
                key: permission,
              },
            },

            select: {
              isAllowed: true,
              scope: true,
            },
          },
        },
      },
    },
  });

  if (!user) {
    return emptyResult();
  }

  const assignment =
    user.employeeAssignments[0] ?? null;

  // مسئولیت ارسالی باید متعلق به همین کاربر و فعال باشد.
  if (assignmentId && !assignment) {
    return emptyResult();
  }

  // مدیر سیستم به تمام مجوزها دسترسی دارد.
  const isSystemAdmin = user.roles.some(
    (userRole) =>
      userRole.role.isSystem &&
      userRole.role.key === "system_admin"
  );

  if (isSystemAdmin) {
    return {
      allowed: true,
      scopes: [DataScope.ALL],
      isSystemAdmin: true,
      activeAssignmentId: assignment?.id ?? null,
    };
  }

  // مجوزهای نقش‌های عمومی کاربر
  const globalGrants = user.roles.flatMap(
    (userRole) => userRole.role.permissions
  );

  // مجوزهای نقش‌های مسئولیت فعال
  const assignmentRoleGrants =
    assignment?.roles.flatMap(
      (assignmentRole) =>
        assignmentRole.role.permissions
    ) ?? [];

  // مجوزهای مستقیم مسئولیت
  const assignmentDirectGrants =
    assignment?.permissions ?? [];

  const grants = [
    ...globalGrants,
    ...assignmentRoleGrants,
    ...assignmentDirectGrants,
  ];

  // منع صریح دسترسی بر اجازه اولویت دارد.
  if (grants.some((grant) => !grant.isAllowed)) {
    return {
      allowed: false,
      scopes: [],
      isSystemAdmin: false,
      activeAssignmentId: assignment?.id ?? null,
    };
  }

  const allowedGrants = grants.filter(
    (grant) => grant.isAllowed
  );

  if (allowedGrants.length === 0) {
    return emptyResult();
  }

  const scopes = [
    ...new Set(
      allowedGrants.map((grant) => grant.scope)
    ),
  ];

  return {
    allowed: true,
    scopes,
    isSystemAdmin: false,
    activeAssignmentId: assignment?.id ?? null,
  };
}

/**
 * بررسی ساده مجوز
 */
export async function hasPermission(
  userId: string,
  permission: string,
  assignmentId?: string
): Promise<boolean> {
  const result = await checkPermission({
    userId,
    permission,
    assignmentId,
  });

  return result.allowed;
}

/**
 * بررسی اجباری مجوز
 */
export async function requirePermission(
  userId: string,
  permission: string,
  assignmentId?: string
): Promise<PermissionResult> {
  const result = await checkPermission({
    userId,
    permission,
    assignmentId,
  });

  if (!result.allowed) {
    throw new PermissionDeniedError();
  }

  return result;
}

/**
 * بررسی دسترسی به اطلاعات متعلق به کاربر
 */
export function canAccessOwnData(
  result: PermissionResult,
  ownerUserId: string,
  currentUserId: string
): boolean {
  if (!result.allowed) {
    return false;
  }

  if (result.scopes.includes(DataScope.ALL)) {
    return true;
  }

  if (result.scopes.includes(DataScope.OWN)) {
    return ownerUserId === currentUserId;
  }

  return false;
}

/**
 * دریافت تمام مجوزهای فعال کاربر
 * برای استفاده در Sidebar و رابط کاربری
 */
export async function getUserPermissions(
  userId: string
): Promise<string[]> {
  const now = new Date();

  const user = await prisma.user.findFirst({
    where: {
      id: userId,
      isActive: true,
    },
    select: {
      roles: {
        where: {
          isActive: true,
          OR: [
            { expiresAt: null },
            { expiresAt: { gt: now } },
          ],
          role: {
            isActive: true,
          },
        },
        select: {
          role: {
            select: {
              key: true,
              isSystem: true,
              permissions: {
                select: {
                  isAllowed: true,
                  permission: {
                    select: {
                      key: true,
                    },
                  },
                },
              },
            },
          },
        },
      },

      employeeAssignments: {
        where: {
          status: "ACTIVE",
          isPrimary: true,
          OR: [
            { endDate: null },
            { endDate: { gt: now } },
          ],
        },
        orderBy: {
          startDate: "desc",
        },
        take: 1,
        select: {
          roles: {
            where: {
              isActive: true,
              OR: [
                { expiresAt: null },
                { expiresAt: { gt: now } },
              ],
              role: {
                isActive: true,
                isSystem: false,
              },
            },
            select: {
              role: {
                select: {
                  permissions: {
                    select: {
                      isAllowed: true,
                      permission: {
                        select: {
                          key: true,
                        },
                      },
                    },
                  },
                },
              },
            },
          },
          permissions: {
            where: {
              isActive: true,
              OR: [
                { expiresAt: null },
                { expiresAt: { gt: now } },
              ],
            },
            select: {
              isAllowed: true,
              permission: {
                select: {
                  key: true,
                },
              },
            },
          },
        },
      },
    },
  });

  if (!user) {
    return [];
  }

  // مدیر سیستم تمام مجوزها را دارد.
  const isSystemAdmin = user.roles.some(
    (item) =>
      item.role.isSystem &&
      item.role.key === "system_admin"
  );

  const allPermissions = await prisma.permission.findMany({
    select: {
      key: true,
    },
  });

  if (isSystemAdmin) {
    return allPermissions.map((item) => item.key);
  }

  const assignment = user.employeeAssignments[0];

  const globalGrants = user.roles.flatMap(
    (item) => item.role.permissions
  );

  const assignmentRoleGrants =
    assignment?.roles.flatMap(
      (item) => item.role.permissions
    ) ?? [];

  const assignmentDirectGrants =
    assignment?.permissions ?? [];

  const grants = [
    ...globalGrants,
    ...assignmentRoleGrants,
    ...assignmentDirectGrants,
  ];

  const deniedPermissions = new Set(
    grants
      .filter((item) => !item.isAllowed)
      .map((item) => item.permission.key)
  );

  const allowedPermissions = new Set(
    grants
      .filter((item) => item.isAllowed)
      .map((item) => item.permission.key)
  );

  return [...allowedPermissions].filter(
    (key) => !deniedPermissions.has(key)
  );
}
